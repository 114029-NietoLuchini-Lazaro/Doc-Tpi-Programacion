package ar.edu.utn.frc.tup.piv.llm.application;
import java.util.UUID; import org.springframework.jdbc.core.JdbcTemplate; import org.springframework.stereotype.Service; import org.springframework.transaction.annotation.Transactional;
@Service public class RecalibrationTriggerService implements RecalibrationScheduler.RecalibrationTrigger {private final JdbcTemplate jdbc;public RecalibrationTriggerService(JdbcTemplate j){jdbc=j;}
 @Transactional public void enqueueMonthlyDue(){jdbc.update("insert into llm.calibration_runs(id,course_id,rubric_version_id,golden_set_version_id,model_deployment_id,reason,created_by_user_id,idempotency_key) select gen_random_uuid(),r.course_id,r.rubric_version_id,r.golden_set_version_id,r.model_deployment_id,'MONTHLY',r.created_by_user_id,gen_random_uuid() from llm.active_calibrations a join llm.calibration_runs r on r.id=a.calibration_run_id where a.activated_at < now()-interval '1 month' and not exists(select 1 from llm.calibration_runs q where q.course_id=r.course_id and q.reason='MONTHLY' and q.state in ('QUEUED','RUNNING'))");}
 @Transactional public void enqueueForModelChange(UUID deployment){jdbc.update("insert into llm.calibration_runs(id,course_id,rubric_version_id,golden_set_version_id,model_deployment_id,reason,created_by_user_id,idempotency_key) select gen_random_uuid(),r.course_id,r.rubric_version_id,r.golden_set_version_id,?,'MODEL_CHANGED',r.created_by_user_id,gen_random_uuid() from llm.active_calibrations a join llm.calibration_runs r on r.id=a.calibration_run_id where not exists(select 1 from llm.calibration_runs q where q.course_id=r.course_id and q.model_deployment_id=? and q.reason='MODEL_CHANGED' and q.state in ('QUEUED','RUNNING'))",deployment,deployment);}}
