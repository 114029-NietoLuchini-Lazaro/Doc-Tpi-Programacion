package ar.edu.utn.frc.tup.piv.llm.application;
import java.time.*; import java.util.*; import java.util.concurrent.*; import org.springframework.stereotype.Component;
/** Temporary in-memory confirmation store; replace with durable token storage with challenges-service. */
@Component public class CalibrationMigrationConfirmation {private final Map<String,Preview> previews=new ConcurrentHashMap<>();public String issue(UUID course,UUID run,Set<UUID> migrable){String token=UUID.randomUUID().toString();previews.put(token,new Preview(course,run,Set.copyOf(migrable),Instant.now().plusSeconds(600)));return token;}public Set<UUID> consume(String token,UUID course,UUID run,Set<UUID> selected){Preview p=previews.remove(token);if(p==null||p.expires().isBefore(Instant.now())||!p.course().equals(course)||!p.run().equals(run)||!p.migrable().containsAll(selected))throw new IllegalStateException("Confirmación de migración inválida");return selected;}private record Preview(UUID course,UUID run,Set<UUID> migrable,Instant expires){}}
