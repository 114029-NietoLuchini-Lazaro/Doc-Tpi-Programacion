package ar.edu.utn.frc.tup.piv.llm.infrastructure.persistence;
import com.fasterxml.jackson.databind.JsonNode; import java.nio.charset.StandardCharsets; import java.security.MessageDigest; import java.util.HexFormat; import java.util.UUID; import org.springframework.jdbc.core.JdbcTemplate; import org.springframework.stereotype.Repository;
@Repository public class CalibrationReproducibilityRepository {private final JdbcTemplate j;public CalibrationReproducibilityRepository(JdbcTemplate j){this.j=j;}public void snapshot(UUID run,JsonNode parameters,String prompt){j.update("update llm.calibration_runs set parameters=cast(? as jsonb),effective_prompt=? where id=? and state='QUEUED'",parameters.toString(),prompt,run);j.update("insert into llm.calibration_run_artifacts(calibration_run_id,artifact_type,content,content_sha256) select r.id,'MODEL_SNAPSHOT',jsonb_build_object('provider',a.provider,'modelId',d.model_id,'modelVersion',d.model_version),? from llm.calibration_runs r join llm.model_deployments d on d.id=r.model_deployment_id join llm.model_adapters a on a.id=d.adapter_id where r.id=? on conflict(calibration_run_id,artifact_type) do nothing",hash(run.toString()),run);}public void artifact(UUID run,String type,JsonNode content){j.update("insert into llm.calibration_run_artifacts(calibration_run_id,artifact_type,content,content_sha256) values(?,?,cast(? as jsonb),?) on conflict(calibration_run_id,artifact_type) do nothing",run,type,content.toString(),hash(content.toString()));}private String hash(String x){try{return HexFormat.of().formatHex(MessageDigest.getInstance("SHA-256").digest(x.getBytes(StandardCharsets.UTF_8)));}catch(Exception e){throw new IllegalStateException(e);}}}
